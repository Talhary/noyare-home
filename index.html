Rust System Call#[tauri::command]
pub async fn get_processes() -> Result<Vec<ProcessInfo>, String> {
    let mut system = System::new_all();
    system.refresh_all();
    
    let processes = system.processes()
        .values()
        .map(|val| ProcessInfo {
            pid: val.pid().as_u32(),
            name: val.name().to_string(),
            cpu: val.cpu_usage(),
            memory: val.memory(),
            path: val.exe().map(|p| p.to_string_lossy().to_string()),
        })
        .collect();
        
    Ok(processes)
}

React Hookexport function useProcessMonitor() {
  const [processes, setProcesses] = useState<Process[]>([]);
  
  useEffect(() => {
    const fetchInterval = setInterval(async () => {
      const data = await invoke<Process[]>("get_processes");
      setProcesses(clusterProcesses(data));
    }, 1000);
    
    return () => clearInterval(fetchInterval);
  }, []);
  
  return { processes };
}
